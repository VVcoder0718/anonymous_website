const express = require('express'); const app = express(); const server = require('http').createServer(app); const io = require('socket.io')(server); app.use(express.static('public')); const rooms = new Map(); io.on('connection', (socket) => { let roomId = null; socket.on('join', (id) => { roomId = id; socket.join(id); if (!rooms.has(id)) { rooms.set(id, { content: '', users: new Set() }); } rooms.get(id).users.add(socket.id); socket.emit('init', rooms.get(id).content); socket.to(id).emit('peers', rooms.get(id).users.size); }); socket.on('update', (content) => { if (roomId && rooms.has(roomId)) { rooms.get(roomId).content = content; socket.to(roomId).emit('update', content); } }); socket.on('disconnect', () => { if (roomId && rooms.has(roomId)) { rooms.get(roomId).users.delete(socket.id); if (rooms.get(roomId).users.size === 0) { rooms.delete(roomId); } } }); }); server.listen(3000);
